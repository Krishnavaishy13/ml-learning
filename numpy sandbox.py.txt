import numpy as np

print("--- Vectors and Dot Products ---")
# Define two 1-dimensional arrays (vectors)
vector_a = np.array([1, 2, 3])
vector_b = np.array([4, 5, 6])

# The dot product multiplies matching index elements and sums them up:
# (1*4) + (2*5) + (3*6) = 4 + 10 + 18 = 32
dot_product = np.dot(vector_a, vector_b)

print(f"Vector A: {vector_a}")
print(f"Vector B: {vector_b}")
print(f"Dot Product: {dot_product}\n")

print("--- Matrix Operations ---")
# Define two 2x2 matrices
matrix_x = np.array([[1, 2], 
                     [3, 4]])

matrix_y = np.array([[5, 6], 
                     [7, 8]])

# Matrix multiplication uses the row-by-column method.
# For example, the top-left result is (1*5) + (2*7) = 19
matrix_multiply = np.dot(matrix_x, matrix_y)

print(f"Matrix Multiplication Result:\n{matrix_multiply}")
